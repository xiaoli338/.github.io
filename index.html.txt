<script>
// 这里是示例数据，你后续可以自行扩充电池档案
const batteryData = {
    "BT001":{model:"12V 100Ah",voltage:"12V",capacity:"100Ah",sn:"BT001",mfg:"2026-01-15"},
    "BT002":{model:"24V 50Ah",voltage:"24V",capacity:"50Ah",sn:"BT002",mfg:"2026-03-22"}
};

function queryBattery(){
    const sn = document.getElementById('sn').value.trim();
    const resDom = document.getElementById('result');
    if(!sn){
        resDom.innerHTML="<p style='color:red'>请输入电池序列号</p>";
        return;
    }
    if(batteryData[sn]){
        const d = batteryData[sn];
        resDom.innerHTML=`
        <div style="padding:16px;background:#f0f7ff;border-radius:8px;">
            <p><strong>电池型号：</strong>${d.model}</p>
            <p><strong>额定电压：</strong>${d.voltage}</p>
            <p><strong>容量：</strong>${d.capacity}</p>
            <p><strong>序列号：</strong>${d.sn}</p>
            <p><strong>生产日期：</strong>${d.mfg}</p>
        </div>`;
    }else{
        resDom.innerHTML="<p style='color:red'>未查询到该电池信息，请检查序列号</p>";
    }
}
</script>
